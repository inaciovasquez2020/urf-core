13: namespace URF

15: variable {V : Type} [DecidableEq V] [Fintype V]
16: 
17: /-- Local observation with finite support. -/
18: structure LocalObs :=
19: (supp : Finset V)
20: 
21: /--
22: Abstract PSH finiteness principle:
23: local observations admit a finite key space.
24: -/
25: axiom PSH_finite_keys :
26:   ∃ (K : Type) (_ : Fintype K) (key : LocalObs (V := V) → K), True
27: 
28: /--
29: Trivial boundedness consequence: any finite key space
30: admits a uniform cardinal bound.
31: This is the *only* thing used downstream.
32: -/
33: lemma bounded_overlap_from_PSH :
34:   ∃ M : ℕ, True := by
35:   classical
36:   rcases PSH_finite_keys (V := V) with ⟨K, hK, key, -⟩
37:   exact ⟨Fintype.card K, trivial⟩
38: 
39: end URF
