6: /-- PSH-style bounded overlap stated directly at the vertex-boundary level. -/
7: axiom PSH_bounded_overlap
8:   {V : Type} [DecidableEq V]
9:   (adj : V → Finset V) (k Δ : ℕ) :
10:   ∃ C : ℕ, ∀ (S : Finset V),
11:     (vertexBoundary adj S).card ≤ C * S.card
